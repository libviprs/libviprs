    /**
     * Tests the whole `hist_equal` chain against libvips on a 16-bit
     * image, which is the end-to-end control on issue #803: the chain is
     * `maplut(hist_norm(hist_cum(hist_find)))`, so narrowing `hist_find`
     * moves every stage and the values have to land where vips lands them.
     * Works by equalising the same values vips was measured on and
     * comparing the output samples and format.
     * Measured on `/opt/homebrew/bin/vips` 8.18.6: `vips hist_equal` of a
     * `ushort` `[4096, 4096, 9]` gives a `USHORT` `[4096, 4096, 1365]`.
     * Input: `[4096, 4096, 9]` -> `[4096, 4096, 1365]`.
     */
    #[test]
    fn hist_equal_matches_vips_on_a_16_bit_image() {
        let eq = gray16(3, 1, &[4096, 4096, 9]).hist_equal();
        assert_eq!((eq.width(), eq.height()), (3, 1));
        assert_eq!(eq.format(), PixelFormat::Gray16);
        assert_eq!(eq.getpoint(0, 0), vec![4096.0]);
        assert_eq!(eq.getpoint(1, 0), vec![4096.0]);
        assert_eq!(eq.getpoint(2, 0), vec![1365.0]);
    }
